# IoTCloud API Overview Base: http://localhost:10000 Protocols: - REST/HTTP → /api/* - WebSocket → /v1/ws - SSE → /v1/events - MQTT/WS → /mqtt Topic format: iotcloud/{projectId}/{deviceId}/{topic} Device tokens authenticate MQTT and REST publishing. Browser clients use HTTPS/WSS automatically.
# Publish telemetry via REST curl -X POST http://localhost:10000/api/device/publish \ -H 'Authorization: Bearer YOUR_DEVICE_TOKEN' \ -H 'content-type: application/json' \ -d '{ "deviceId": "YOUR_DEVICE_ID", "topic": "telemetry", "data": {"temperature": 28.4} }' Response 200: { "ok": true } Response 403: { "error": "device token required" }
# WebSocket event stream const ws = new WebSocket( ws://localhost:10000/v1/ws?token=YOUR_DEVICE_TOKEN ); ws.onopen = () => console.log('connected'); ws.onmessage = e => console.log(JSON.parse(e.data)); ws.onerror = e => console.error(e); ws.onclose = e => console.log('closed', e.code); Publish: ws.send(JSON.stringify({ action: 'publish', topic: 'telemetry', data: { temperature: 28.4 } }));
# MQTT over WebSocket Topic: iotcloud/{projectId}/{deviceId}/{topic} Example: iotcloud/PROJECT_ID/DEVICE_ID/telemetry Python (paho-mqtt): client.connect(localhost, 443, 60) client.publish( 'iotcloud/PROJECT_ID/DEVICE_ID/telemetry', json.dumps({'temperature': 28.4}) )
# AI Context for IoTCloud troubleshooting Project ID: YOUR_PROJECT_ID Device ID: YOUR_DEVICE_ID Device Token: YOUR_DEVICE_TOKEN Base URL: http://localhost:10000 WebSocket: ws://localhost:10000/v1/ws?token=DEVICE_TOKEN SSE: http://localhost:10000/v1/events?token=DEVICE_TOKEN MQTT/WS: ws://localhost:10000/mqtt Topic: iotcloud/PROJECT_ID/DEVICE_ID/telemetry # Interconnection # REST POST /api/device/publish → broker.publish() → mqtt.message event # WebSocket /v1/ws publish action → broker.publish() → mqtt.message event # MQTT /mqtt publish → broker.publish() → mqtt.message event # All paths converge at the broker and emit to WS + SSE clients.
