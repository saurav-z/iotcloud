# Publish telemetry via REST curl -X POST https://iotcloud-api-a04c.onrender.com/api/device/publish \ -H 'Authorization: Bearer DEVICE_TOKEN' \ -H 'content-type: application/json' \ -d '{ "deviceId": "DEVICE_ID", "topic": "telemetry", "data": {"temperature": 28.4} }' Response 200: { "ok": true } Response 403: { "error": "device token required" }
